
```python
from uuid import NAMESPACE_URL, uuid5

from qdrant_client import QdrantClient, models

from embedding_io import load_chunks, load_vectors, load_model

  

# 준비 단계에서 만든 벡터의 행 수가 청크 수와 같은지 확인해요.

# 검색 결과와 본문을 잘못 연결하지 않으려면 두 자료의 순서를 바꾸면 안 돼요.

chunks = load_chunks()

vectors = load_vectors(chunks)

print(len(chunks), vectors.shape)

print(chunks[0]["document_id"], chunks[0]["start"])

  

# 로컬 메모리 Collection을 만들고 벡터의 차원과 유사도 기준을 지정해요.

# 아직 Upsert 전이므로 처음 Point 수는 0이에요.

client = QdrantClient(":memory:")

collection = "campus_saved_e5"

client.create_collection(

    collection,

    vectors_config=models.VectorParams(

        size=int(vectors.shape[1]),

        distance=models.Distance.COSINE,

    ),

)

print(client.count(collection).count)

  

# 모든 청크를 ID, 벡터, payload가 있는 Point로 변환해요.

# NumPy 벡터는 Qdrant 입력에 맞게 일반 목록으로 바꿔요.

points = []

for chunk, vector in zip(chunks, vectors):

    point = models.PointStruct(

        id=str(uuid5(NAMESPACE_URL, chunk["chunk_id"])),

        vector=vector.tolist(),

        payload=chunk,

    )

    points.append(point)

  

# wait=True로 등록 완료를 기다린 뒤 개수를 세어 누락을 확인해요.

client.upsert(collection, points=points, wait=True)

print(client.count(collection).count)

  

# 첫 Point를 ID로 다시 읽어 본문과 위치가 입력 자료와 같은지 확인해요.

# 이 조회에는 임베딩이나 검색 질문이 필요하지 않아요.

first = client.retrieve(

    collection,

    ids=[points[0].id],

    with_payload=True,

)[0]

print(first.payload["document_id"], first.payload["start"])

  

# 문서 벡터를 만든 것과 같은 모델로 질문을 임베딩해요.

# 문서의 passage 대신 query 접두사를 쓰되 정규화 설정은 동일하게 맞춰요.

model = load_model()

question = "출장 숙박 비용 한도와 필요한 증빙은?"

query = model.encode(

    "query: " + question,

    normalize_embeddings=True,

)

print(query.shape)

  

# payload 안의 metadata.status 경로에 현재 상태 조건을 걸어요.

# 벡터가 비슷해도 폐기된 규정이면 검색 후보에서 제외돼요.

current_filter = models.Filter(

    must=[

        models.FieldCondition(

            key="metadata.status",

            match=models.MatchValue(value="current"),

        )

    ]

)

  

# 코사인 점수가 높은 현재 청크 세 개를 받아요.

# with_payload=True가 있어야 점수뿐 아니라 답변 근거와 출처까지 사용할 수 있어요.

hits = client.query_points(

    collection,

    query=query.tolist(),

    limit=3,

    query_filter=current_filter,

    with_payload=True,

).points

rows = [{**hit.payload, "score": float(hit.score)} for hit in hits]

for rank, row in enumerate(rows, 1):

    print(rank, row["document_id"], row["start"],

          round(row["score"], 4))

  

print(rows[0]["text"])

print(rows[0]["metadata"]["source_url"])

  

from saved_index import SavedDenseIndex

  

# 같은 자료와 질문을 도우미 클래스로 실행해 직접 작성한 검색과 순서가 같은지 확인해요.

# 클래스가 검색 원리를 바꾸는 것이 아니라 방금 작성한 절차를 묶은 것임을 보여 줘요.

index = SavedDenseIndex(chunks, vectors, model)

again = index.search(question, k=3)

print([r["chunk_id"] for r in rows] ==

      [r["chunk_id"] for r in again])

index.close()

  

# TODO: 본문의 실습 코드를 이 줄 아래, 연결 종료 코드 위에 작성하세요.

  

client.close()
```
